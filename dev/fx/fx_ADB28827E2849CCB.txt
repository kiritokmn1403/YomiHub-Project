			            menuToggle.classList.toggle('toggled');
			
			            // Toggle classes for el-group

			        });
			    }
			
			    // Mobile search toggle functionality
			    const searchToggle = document.querySelector('.for-mobile.search-toggle');
			    const headerElements = document.querySelector('.header-elements');
			    const searchElement = document.querySelector('#search');
			
			    if (searchToggle && headerElements && searchElement) {

			            // Toggle classes for search toggle
			            searchToggle.classList.toggle('toggled');
			
			            // Toggle classes for header-elements
			            headerElements.classList.toggle('simple');
			
			            // Toggle classes for search element
			            searchElement.classList.toggle('toggled');
			        });
			    }
			
			    // Get all dropdown triggers
			    const dropdownTriggers = document.querySelectorAll('.season-dropdown > a');
			
			    dropdownTriggers.forEach(function(trigger) {
			        const dropdown = trigger.parentElement.querySelector('.dropdown-menu');
			
			        // Add click event to toggle dropdown
			        trigger.addEventListener('click', function(e) {
			            e.preventDefault();
			
			            // Close all other dropdowns
			            document.querySelectorAll('.dropdown-menu').forEach(function(menu) {
			                if (menu !== dropdown) {
			                    menu.classList.remove('show');
			                }
			            });
			
			            // Toggle current dropdown
			            dropdown.classList.toggle('show');
			        });
			    });
			
			    // Close dropdowns when clicking outside
			    document.addEventListener('click', function(e) {
			        if (!e.target.closest('.season-dropdown')) {
			            document.querySelectorAll('.dropdown-menu').forEach(function(menu) {
			                menu.classList.remove('show');
			            });
			        }
			    });
			
			    // Close dropdowns on escape key
			    document.addEventListener('keydown', function(e) {
			        if (e.key === 'Escape') {
			            document.querySelectorAll('.dropdown-menu').forEach(function(menu) {
			                menu.classList.remove('show');
			            });
			        }
			    });
			
			    // Search functionality
			    const searchInput = document.querySelector('#main-search');
			    const searchIcon = document.querySelector('.search-icon');

			        searchInput.addEventListener('keydown', function(e) {
			            const suggestions = document.querySelectorAll(

			            if (e.key === 'ArrowDown') {
			                e.preventDefault();
			                selectedSuggestionIndex = Math.min(selectedSuggestionIndex + 1, suggestions.length -
			                    1);
			                updateSelectedSuggestion(suggestions);
			            } else if (e.key === 'ArrowUp') {
			                e.preventDefault();
			                selectedSuggestionIndex = Math.max(selectedSuggestionIndex - 1, -1);
			                updateSelectedSuggestion(suggestions);
			            } else if (e.key === 'Enter') {
			                e.preventDefault();
			                if (selectedSuggestionIndex >= 0 && suggestions[selectedSuggestionIndex]) {
			                    suggestions[selectedSuggestionIndex].click();

			                    performSearch();
			                }

			                hideSuggestions();
			                selectedSuggestionIndex = -1;

			                hideSuggestions();

			        document.addEventListener('click', function(e) {
			            if (!searchInput.contains(e.target) && !suggestionsContainer.contains(e.target)) {

			        });
			
			        // Show suggestions when focusing on input
			        searchInput.addEventListener('focus', function() {
			            const query = this.value.trim();
			            if (query.length >= 2) {

			            }
			        });

			    // Search function
			    function performSearch() {
			        const searchInput = document.querySelector('#main-search');
			        if (searchInput) {
			            const query = searchInput.value.trim();
			
			            if (query.length > 0) {
			                hideSuggestions();
			                // Redirect to search results page
			                const searchUrl = 'https://hotphim.org/?s=' + encodeURIComponent(query);
			                window.location.href = searchUrl;
			            }
			        }
			    }
			
			    // Fetch search suggestions
			    function fetchSearchSuggestions(query) {
			        const suggestionsContainer = document.querySelector('#search-suggestions');
			
			        // Show loading state
			        suggestionsContainer.innerHTML = '<div class="suggestion-item loading">Đang tìm kiếm...</div>';
			        suggestionsContainer.style.display = 'block';

			        formData.append('nonce', '9c0845c61a');

			                method: 'POST',
			                body: formData
			            })
			            .then(response => response.json())
			            .then(data => {
			                if (data.success && data.data.length > 0) {
			                    displaySuggestions(data.data);
			                } else {
			                    suggestionsContainer.innerHTML =
			                        '<div class="suggestion-item no-results">Không tìm thấy gợi ý</div>';
			                }
			            })
			            .catch(error => {
			                console.error('Search suggestions error:', error);
			                hideSuggestions();
			            });
			    }
			
			    // Display search suggestions
			    function displaySuggestions(suggestions) {
			        const suggestionsContainer = document.querySelector('#search-suggestions');
			        let html = '';
			
			        // Reset selected index
			        selectedSuggestionIndex = -1;

			                    <i class="fa-solid ${icon}"></i>
			                    <span>${suggestion.title}</span>
			                    ${suggestion.type === 'category' ? '<small class="suggestion-type">Thể loại</small>' : ''}
			                </div>`;

			        suggestionItems.forEach(item => {

			                const query = this.getAttribute('data-query');
			                const url = this.getAttribute('data-url');
			
			                if (url && url !== 'undefined') {

			                    window.location.href = url;
			                } else {
			                    // Perform search with the query
			                    const searchInput = document.querySelector('#main-search');
			                    searchInput.value = query;
			                    performSearch();
			                }
			            });
			        });

			        const suggestionsContainer = document.querySelector('#search-suggestions');

    jQuery(document).ready(function($) {
        $(document).on('click', '.load-more-btn', function(e) {
            e.preventDefault();

            var $btn = $(this);
            var $btnText = $btn.find('.btn-text');
            var $loadingText = $btn.find('.loading-text');
            var postId = $btn.data('post-id');
            var offset = $btn.data('offset');
            var isLoading = $btn.data('loading');

            // Ngăn chặn nhiều lần nhấp chuột
            if (isLoading) {
                return;
            }

            // Đặt trạng thái tải
            $btn.data('loading', true);
            $btnText.hide();
            $loadingText.show();

            // AJAX request
            $.ajax({
                url: comment_ajax.ajax_url,
                type: 'POST',
                data: {
                    action: 'load_more_comments',
                    post_id: postId,
                    offset: offset,
                    nonce: comment_ajax.nonce
                },
                success: function(response) {
                    if (response.success) {
                        // Chèn bình luận mới trước nút tải thêm
                        $btn.closest('.more').before(response.data.html);

                        // Cập nhật độ lệch
                        $btn.data('offset', response.data.next_offset);

                        // Ẩn nút tải thêm nếu không còn bình luận nào nữa
                        if (!response.data.has_more) {
                            $btn.closest('.more').hide();
                        }
                    } else {
                        console.error('Error loading comments:', response.data);
                        alert('Có lỗi xảy ra khi tải bình luận. Vui lòng thử lại.');
                    }
                },
                error: function(xhr, status, error) {
                    console.error('AJAX error:', error);
                    alert('Có lỗi xảy ra khi tải bình luận. Vui lòng thử lại.');
                },
                complete: function() {
                    // Đặt lại trạng thái tải
                    $btn.data('loading', false);
                    $btnText.show();
                    $loadingText.hide();
                }
            });
        });
    });

    <script>
    document.addEventListener('DOMContentLoaded', function() {
        const ratingModal = document.getElementById('ratingModal');
        const ratingTrigger = document.querySelector('.rating-popup-trigger');
        const closeModal = document.getElementById('closeRatingModal');
        const cancelRating = document.getElementById('cancelRating');
        const submitRating = document.getElementById('submitRating');
        const ratingItems = document.querySelectorAll('.rate-emo .item-v');
        const ratingComment = document.getElementById('ratingComment');
        const currentRatingDisplay = document.getElementById('currentRating');
        const movieTitle = document.getElementById('movieTitle');

        let selectedRating = 0;

        // Mở modal khi click vào đánh giá
        if (ratingTrigger) {
            ratingTrigger.addEventListener('click', function() {
                ratingModal.style.display = 'block';
                document.body.style.overflow = 'hidden';
                ratingModal.classList.add('show');
            });
        }

        // Function đóng modal
        function closeModalFunc() {
            ratingModal.style.display = 'none';
            document.body.style.overflow = 'auto';
            ratingModal.classList.remove('show');
            selectedRating = 0;
            if (ratingComment) ratingComment.value = '';
            updateRatingDisplay();
        }

        if (closeModal) closeModal.addEventListener('click', closeModalFunc);
        if (cancelRating) cancelRating.addEventListener('click', closeModalFunc);

        // Đóng modal khi click bên ngoài
        if (ratingModal) {
            ratingModal.addEventListener('click', function(e) {
                if (e.target === ratingModal) {
                    closeModalFunc();
                }
            });
        }

        // Xử lý chọn đánh giá
        ratingItems.forEach(item => {
            item.addEventListener('click', function() {
                selectedRating = parseInt(this.dataset.rating);
                updateRatingDisplay();
                highlightSelectedRating();
            });
        });

        // Cập nhật hiển thị đánh giá
        function updateRatingDisplay() {
            if (currentRatingDisplay) {
                if (selectedRating > 0) {
                    currentRatingDisplay.textContent = selectedRating + '.0';
                } else {
                    currentRatingDisplay.textContent =
                        '0';
                }
            }
        }

        // Highlight đánh giá được chọn
        function highlightSelectedRating() {
            ratingItems.forEach(item => {
                item.classList.remove('active');
                if (parseInt(item.dataset.rating) === selectedRating) {
                    item.classList.add('active');
                }
            });
        }

        // Gửi đánh giá
        if (submitRating) {
            submitRating.addEventListener('click', function() {
                if (selectedRating === 0) {
                    alert('Vui lòng chọn đánh giá trước khi gửi!');
                    return;
                }

                const formData = new FormData();
                formData.append('action', 'submit_movie_rating');
                formData.append('post_id', '3711');
                formData.append('rating', selectedRating);
                formData.append('comment', ratingComment ? ratingComment.value : '');
                formData.append('rating_nonce',
                    'd22c294854');

                // Vô hiệu hóa nút gửi
                submitRating.disabled = true;
                submitRating.textContent = 'Đang gửi...';

                fetch('https://hotphim.org/wp-admin/admin-ajax.php', {
                        method: 'POST',
                        body: formData
                    })
                    .then(response => response.json())
                    .then(data => {
                        if (data.success) {
                            // Cập nhật hiển thị đánh giá
                            const ratingDisplay = document.querySelector('.ro-rating .point');
                            const ratingCount = document.querySelector('.ro-rating .a-rate');
                            const modalRating = document.getElementById('currentRating');
                            const modalCount = document.querySelector('#currentRating')
                                .nextElementSibling;

                            if (ratingDisplay) ratingDisplay.textContent = data.data.avg_rating;
                            if (ratingCount) ratingCount.textContent = data.data.rating_count +
                                ' đánh giá';
                            if (modalRating) modalRating.textContent = data.data.avg_rating;
                            if (modalCount) modalCount.textContent = '/ ' + data.data
                                .rating_count + ' lượt đánh giá';

                            // Hiển thị thông báo thành công
                            alert('Cảm ơn bạn đã đánh giá phim!');

                            // Đóng modal
                            closeModalFunc();
                        } else {
                            alert('Có lỗi xảy ra: ' + (data.data || 'Không thể gửi đánh giá'));
                        }
                    })
                    .catch(error => {
                        console.error('Error:', error);
                        alert('Có lỗi xảy ra khi gửi đánh giá');
                    })
                    .finally(() => {
                        // Kích hoạt lại nút gửi
                        submitRating.disabled = false;
                        submitRating.textContent = 'Gửi đánh giá';
                    });
            });
        }
    });
    </script>
